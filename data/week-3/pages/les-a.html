ORDER BY en GROUP BY

ORDER BY

Gebruik ORDER BY om resultaten te sorteren. Met DESC zet je hoogste waarden bovenaan, met ASC laagste eerst.

SELECT product, omzet
FROM verkopen
ORDER BY omzet DESC;

GROUP BY

GROUP BY bundelt rijen met dezelfde waarde. Daarna kun je aggregaties toepassen zoals SUM(), AVG() en COUNT().

SELECT land, SUM(omzet) AS totale_omzet
FROM verkopen
GROUP BY land;
